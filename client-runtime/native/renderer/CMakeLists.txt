cmake_minimum_required(VERSION 3.20)
project(ArcadiumResidentScene LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_VISIBILITY_PRESET hidden)
option(ARC_PRESENT_HOSTED_EXPERIMENT "Compare compositor-owned layer hosting with the existing backing control" OFF)
option(ARC_BUILD_PRESENTATION_TESTING "Build the isolated drawable-readback QA library" OFF)
if(ARC_PRESENT_HOSTED_EXPERIMENT)
    add_compile_definitions(ARC_PRESENT_HOSTED_EXPERIMENT=1)
endif()
if(APPLE)
    enable_language(OBJC)
    if(NOT CMAKE_OSX_DEPLOYMENT_TARGET)
        set(CMAKE_OSX_DEPLOYMENT_TARGET "12.0")
    endif()
    add_library(arcadium_renderer_metal SHARED arc_scene_metal.m arc_scene_validate.c arc_scene_passes.c arc_present_macos.m arc_present_validate.c arc_present_scene_validate.c)
    target_compile_options(arcadium_renderer_metal PRIVATE "$<$<COMPILE_LANGUAGE:OBJC>:-fobjc-arc;-fblocks>")
    target_link_libraries(arcadium_renderer_metal PRIVATE "-framework Foundation" "-framework Metal"
        "-framework Cocoa" "-framework QuartzCore" "-framework CoreVideo" "-framework IOSurface" "-framework OpenGL")
    if(ARC_BUILD_PRESENTATION_TESTING)
        add_library(arcadium_renderer_metal_readback_qa SHARED tests/scene_present_testing.m arc_scene_metal.m arc_scene_validate.c arc_scene_passes.c arc_present_validate.c arc_present_scene_validate.c)
        target_compile_options(arcadium_renderer_metal_readback_qa PRIVATE "$<$<COMPILE_LANGUAGE:OBJC>:-fobjc-arc;-fblocks>" -Wall -Wextra -Werror -Wno-deprecated-declarations)
        target_link_libraries(arcadium_renderer_metal_readback_qa PRIVATE "-framework Foundation" "-framework Metal"
            "-framework Cocoa" "-framework QuartzCore" "-framework CoreVideo" "-framework IOSurface" "-framework OpenGL")
    endif()
else()
    add_library(arcadium_renderer_metal SHARED arc_scene_unsupported.c arc_scene_validate.c arc_scene_passes.c arc_present_unsupported.c arc_present_validate.c arc_present_scene_validate.c)
endif()
if(MSVC)
    target_compile_options(arcadium_renderer_metal PRIVATE /W4 /WX)
else()
    target_compile_options(arcadium_renderer_metal PRIVATE -Wall -Wextra -Werror)
endif()
add_executable(arcadium_scene_contract_test tests/scene_contract_test.c arc_scene_validate.c)
enable_testing()
add_test(NAME scene_contract COMMAND arcadium_scene_contract_test)
add_executable(arcadium_scene_pass_contract_test tests/scene_pass_contract_test.c arc_scene_passes.c arc_scene_validate.c)
add_test(NAME scene_pass_contract COMMAND arcadium_scene_pass_contract_test)
add_executable(arcadium_metal_lease_contract_test tests/metal_lease_contract_test.c)
target_link_libraries(arcadium_metal_lease_contract_test PRIVATE arcadium_renderer_metal)
add_test(NAME metal_lease_contract COMMAND arcadium_metal_lease_contract_test)
add_executable(arcadium_presentation_contract_test tests/presentation_contract_test.c arc_present_validate.c)
if(CMAKE_C_COMPILER_ID STREQUAL "MSVC")
    target_compile_options(arcadium_presentation_contract_test PRIVATE /experimental:c11atomics)
endif()
add_test(NAME presentation_contract COMMAND arcadium_presentation_contract_test)
add_executable(arcadium_scene_present_contract_test tests/scene_present_contract_test.c arc_present_scene_validate.c)
target_link_libraries(arcadium_scene_present_contract_test PRIVATE arcadium_renderer_metal)
add_test(NAME scene_present_contract COMMAND arcadium_scene_present_contract_test)
if(NOT APPLE)
    add_executable(arcadium_presentation_unsupported_test tests/presentation_unsupported_test.c)
    target_link_libraries(arcadium_presentation_unsupported_test PRIVATE arcadium_renderer_metal)
    add_test(NAME presentation_unsupported COMMAND arcadium_presentation_unsupported_test)
endif()
if(APPLE)
    add_executable(arcadium_metal_scene_test tests/metal_scene_test.m)
    target_compile_options(arcadium_metal_scene_test PRIVATE -fobjc-arc -Wall -Wextra -Werror)
    target_link_libraries(arcadium_metal_scene_test PRIVATE arcadium_renderer_metal "-framework Foundation" "-framework Metal")
    add_test(NAME metal_resident_scene COMMAND arcadium_metal_scene_test "${CMAKE_CURRENT_SOURCE_DIR}/tests/reference_scene.metal")
    set_tests_properties(metal_resident_scene PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 30)
    add_executable(arcadium_metal_passes_test tests/metal_passes_test.m)
    target_compile_options(arcadium_metal_passes_test PRIVATE -fobjc-arc -Wall -Wextra -Werror -Wno-deprecated-declarations)
    target_link_libraries(arcadium_metal_passes_test PRIVATE arcadium_renderer_metal "-framework Cocoa" "-framework OpenGL" "-framework Metal")
    add_test(NAME metal_ordered_passes COMMAND arcadium_metal_passes_test "${CMAKE_CURRENT_SOURCE_DIR}/tests/reference_passes.metal" "${CMAKE_CURRENT_BINARY_DIR}/metal-pass-result.json")
    set_tests_properties(metal_ordered_passes PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 45)
    add_executable(arcadium_metal_lease_test tests/metal_lease_test.m)
    target_compile_options(arcadium_metal_lease_test PRIVATE -fobjc-arc -fblocks -Wall -Wextra -Werror)
    target_link_libraries(arcadium_metal_lease_test PRIVATE arcadium_renderer_metal "-framework Foundation" "-framework Metal")
    add_test(NAME metal_target_leases COMMAND arcadium_metal_lease_test "${CMAKE_CURRENT_SOURCE_DIR}/tests/reference_scene.metal" "${CMAKE_CURRENT_BINARY_DIR}/metal-lease-result.json")
    set_tests_properties(metal_target_leases PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 45)
    add_executable(arcadium_metal_presentation_test tests/metal_presentation_test.m arc_scene_metal.m arc_scene_validate.c arc_scene_passes.c arc_present_validate.c arc_present_scene_validate.c)
    target_compile_options(arcadium_metal_presentation_test PRIVATE "$<$<COMPILE_LANGUAGE:OBJC>:-fobjc-arc;-fblocks>" -Wall -Wextra -Werror -Wno-deprecated-declarations)
    target_link_libraries(arcadium_metal_presentation_test PRIVATE "-framework Foundation" "-framework Metal"
        "-framework Cocoa" "-framework QuartzCore" "-framework CoreVideo" "-framework IOSurface" "-framework OpenGL")
    add_test(NAME metal_current_frame_presentation COMMAND arcadium_metal_presentation_test)
    set_tests_properties(metal_current_frame_presentation PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 45)
endif()
